          parent=${OBS_PARENT:--}; start=$(date +%s)
          status=0; OBS_TRACE=$trace OBS_PARENT=$sid "$@" || status=$?
          end=$(date +%s); label=ok or exit=$status
          echo "$trace $sid $parent $start $((end-start)) $label $name" >> "$spans"
          metrics span.total (+ span.errors); exit "$status"
