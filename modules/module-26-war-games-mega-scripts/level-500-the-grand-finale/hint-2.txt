Days since 1970 from year/month/day ("days from civil"):
  (( m <= 2 )) && y=$((y - 1))
  era=$(( (y >= 0 ? y : y - 399) / 400 ));  yoe=$((y - era * 400))
  mp=$(( m > 2 ? m - 3 : m + 9 ));  doy=$(( (153 * mp + 2) / 5 + d - 1 ))
  doe=$(( yoe * 365 + yoe / 4 - yoe / 100 + doy ));  days=$(( era * 146097 + doe - 719468 ))
