  refill() {   # CLIENT TS
    local c=$1 ts=$2 added
    if (( ts > last[$c] )); then
      added=$(( (ts - last[$c]) / spt[$c] ))
      tokens[$c]=$(( tokens[$c] + added ))
      last[$c]=$(( last[$c] + added * spt[$c] ))
    fi
    if (( tokens[$c] >= cap[$c] )); then tokens[$c]=${cap[$c]}; last[$c]=$ts; fi
  }
