Skeleton:

  [[ $# -eq 2 ]] || usage;  app=$1; version=$2
  [[ $version =~ ^[0-9]+\.[0-9]+\.[0-9]+$ ]] || { echo "error: bad version '$version'" >&2; exit 2; }
  lock=state/$app.lock; mkdir -p state
  [[ -e $lock ]] && { ...; exit 7; }
  : > "$lock"; trap 'rm -f "$lock"' EXIT
  previous=none; [[ -f state/$app.version ]] && previous=$(<state/$app.version)

  run_stage() { local name=$1; shift
    if "$@" >/dev/null 2>&1; then echo "stage $name: ok"; log ok; else echo "stage $name: FAILED"; log fail; return 1; fi; }
  verify()   { for attempt in 1 2 3; do harbor-health ... && { echo "stage verify: ok (attempt $attempt)"; return 0; }; (( attempt < 3 )) && sleep 5; done; echo "stage verify: FAILED"; return 1; }
  rollback() { [[ $previous == none ]] && { echo "error: ..." >&2; exit 6; }; harbor-deploy "$app" "$previous" ...; echo "rolled back to $previous"; }

  build -> exit 3, test -> exit 4, then: if ! run_stage deploy ... || ! verify; then rollback; ...; exit 5; fi
  printf '%s\n' "$version" > state/$app.version; echo "deployed ..."
