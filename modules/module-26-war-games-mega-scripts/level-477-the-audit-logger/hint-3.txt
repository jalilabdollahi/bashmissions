  checksum() { local sum _; read -r sum _ < <(printf '%s' "$1|$2|$3|$4|$5|$6" | cksum); echo "$sum"; }

  walk() {                        # sets entries, seg_count, last_seq, last_sum, bad
    prev=0 expect=1
    for f in seg-*.log then current.log; do lineno=0
      while IFS='|' read -r seq sum ts actor action detail || [[ -n $seq ]]; do
        lineno++ ; format check -> bad="${f##*/} line $lineno (bad format)"; return 1
        seq != expect  -> (sequence gap) ;  sum != checksum(...) -> (checksum mismatch)
        prev=$sum; expect++ ; entries++
      done < "$f"
    done
    last_seq=$((expect - 1)); last_sum=$prev
  }
