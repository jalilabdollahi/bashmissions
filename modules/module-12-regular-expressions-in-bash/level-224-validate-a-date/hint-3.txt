  year=$(( 10#${BASH_REMATCH[1]} ))
  month=$(( 10#${BASH_REMATCH[2]} ))
  day=$(( 10#${BASH_REMATCH[3]} ))
  (( month < 1 || month > 12 )) -> "bad month"

  case $month in
    4|6|9|11) max=30 ;;
    2) leap if year % 400 == 0, or (year % 4 == 0 and year % 100 != 0): max=29, else 28 ;;
    *) max=31 ;;
  esac
  (( day < 1 || day > max )) -> "bad day", otherwise "ok"
