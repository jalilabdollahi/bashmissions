  re='^[[:alnum:]._%+-]+@[[:alnum:].-]+\.[[:alpha:]]{2,}$'
  while IFS= read -r addr || [[ -n $addr ]]; do
    if [[ $addr =~ $re ]]; then
      echo "OK $addr"
    else
      echo "BAD $addr"
    fi
  done
