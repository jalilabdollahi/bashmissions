  lineno=0
  while IFS= read -r line || [[ -n $line ]]; do
    lineno=$(( lineno + 1 ))
    [[ $line =~ ^[[:space:]]*(#.*)?$ ]] && continue
    if [[ $line =~ ^[[:space:]]*([A-Za-z_][A-Za-z0-9_]*)[[:space:]]*=(.*)$ ]]; then
      key=${BASH_REMATCH[1]}; value=${BASH_REMATCH[2]}
    else
      echo "skipped line $lineno" >&2; continue
    fi
    # trim both ends of value, then strip one pair of " or ' with ${value:1:${#value}-2}
    export "$key=$value"
  done < "$1"
  echo "name=${SITE_NAME:-(unset)}"     # ...and the other two
