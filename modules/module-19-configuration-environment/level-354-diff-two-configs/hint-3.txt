  load() {
    local file=$1 key val
    local -n target=$2
    while IFS='=' read -r key val || [[ -n $key ]]; do
      # trim key and val, skip empty keys and comments
      target[$key]=$val
    done < "$file"
  }
  declare -A old new
  load "$1" old; load "$2" new
  while IFS= read -r key; do
    [[ -n $key ]] || continue
    if [[ -z ${old[$key]+set} ]]; then      # only in new: "+ ..."
    elif [[ -z ${new[$key]+set} ]]; then    # only in old: "- ..."
    elif [[ ${old[$key]} != "${new[$key]}" ]]; then   # "~ key: old -> new"
    fi
  done < <(printf '%s\n' "${!old[@]}" "${!new[@]}" | sort -u)
