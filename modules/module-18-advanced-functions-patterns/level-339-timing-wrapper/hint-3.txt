  timed() {
    local start end rc=0 ms
    start=${EPOCHREALTIME/./}
    "$@" || rc=$?
    end=${EPOCHREALTIME/./}
    ms=$(( (end - start) / 1000 ))
    printf 'timed: %s took %d.%03ds\n' "$*" $(( ms / 1000 )) $(( ms % 1000 )) >&2
    return "$rc"
  }
  # no arguments: usage, exit 2
  status=0
  timed "$@" || status=$?
  exit "$status"
