  allowed=(restart reload status)
  is_allowed() { ...exact match loop over "${allowed[@]}"...; }

  refused=0
  while read -r action target; do
    [[ -z $action ]] && continue
    if ! is_allowed "$action"; then
      echo "refused: '$action' is not an allowed action" >&2
      refused=1
      continue
    fi
    "do_$action" "$target"
  done
  # exit 3 if refused is 1
