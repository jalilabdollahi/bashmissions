  retry() {
    local max=$1 attempt=1 rc=0
    shift
    while true; do
      rc=0
      "$@" || rc=$?
      # rc == 0: return 0
      echo "retry: attempt $attempt/$max failed (status $rc)" >&2
      # attempt >= max: print "retry: giving up" and return "$rc"
      sleep "${RETRY_DELAY:-0}"
      attempt=$(( attempt + 1 ))
    done
  }
  # validate $# >= 2, $1 numeric and >= 1; else usage, exit 2
  status=0
  retry "$@" || status=$?
  exit "$status"
