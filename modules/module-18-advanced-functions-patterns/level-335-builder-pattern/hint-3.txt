  declare -A spec=()
  vm_new() { spec=([name]=$1 [cpus]=1 [ram_gb]=2 [disk_gb]=20 [region]=eu-north); }
  in_range() { [[ $1 =~ ^[0-9]+$ ]] && (( $1 >= $2 && $1 <= $3 )); }
  with_cpus() {
    if ! in_range "$1" 1 16; then echo "with_cpus: invalid value '$1'" >&2; return 1; fi
    spec[cpus]=$1
  }
  # with_ram_gb, with_disk_gb, with_region: same shape
  build() { need=$(( 2 * spec[cpus] )); ...compare, then print the five lines...; }
  vm_new "$1"; shift
  for arg in "$@"; do
    key=${arg%%=*}; value=${arg#*=}
    case $key in cpus) with_cpus "$value" || exit 2 ;; ... *) unknown option ;; esac
  done
  build || exit 2
