  declare -A count=()
  total=0
  while IFS= read -r path; do
    name=${path##*/}
    stem=${name%.*}
    ext=${name##*.}
    if [[ $name == *.* && -n $stem && -n $ext ]]; then ext=${ext,,}; else ext="(none)"; fi
    count[$ext]=$(( ${count[$ext]:-0} + 1 ))
    total=$(( total + 1 ))
  done < <(find "$folder" -type f)
