  awk '
    { sub(/\r$/, "") }
    $0 == "" { next }
    {
      total++
      if ($4 ~ /^[1-5][0-9][0-9]$/) cls = substr($4, 1, 1) "xx"
      else cls = "other"
      count[cls]++
    }
    END {
      n = split("1xx 2xx 3xx 4xx 5xx other", names, " ")
      for (i = 1; i <= n; i++) {
        c = count[names[i]] + 0
        # pct: 100 * c / total, but 0 when total is 0
        printf "%s %d %.1f%%\n", names[i], c, pct
      }
      # printf the total line
    }
  ' "$1"
