  limit=${2:-3}
  [[ $limit =~ ^[1-9][0-9]*$ ]] || { echo "usage: solution.sh LOG [N]" >&2; exit 2; }

  awk '
    { sub(/\r$/, "") }
    NF < 3 { next }
    {
      requests[$2]++
      # add $3 to bytes[$2] only when it is a number; make sure bytes[$2] exists
    }
    END { for (c in requests) printf "%s %d %.0f\n", c, requests[c], bytes[c] }
  ' "$1" |
    sort -k3,3nr -k1,1 |
    awk -v limit="$limit" 'NR <= limit { printf "%d. %s requests=%d bytes=%s\n", NR, $1, $2, $3 }'
