  [[ $# -ge 3 ]] || { echo "usage: solution.sh LOG FROM TO" >&2; exit 2; }
  [[ $2 > $3 ]] && { echo "error: FROM is after TO" >&2; exit 1; }

  awk -v from="$2" -v to="$3" '
    { sub(/\r$/, "") }
    $0 == "" { next }
    $1 !~ /^[0-9][0-9][0-9][0-9]-[0-9][0-9]-[0-9][0-9]T[0-9][0-9]:[0-9][0-9]:[0-9][0-9]Z$/ {
      # report skipped line NR on stderr
      next
    }
    # pattern only: the lower bound is inclusive, the upper one is not
  ' "$1"

With `set -e`, write the second check as an if statement: `a && b` as a last command can end the script.
