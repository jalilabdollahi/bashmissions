  if ! pg_dump --no-owner "$db" | gzip > "$partial"; then
    rm -f "$partial"; echo "dump failed for $db" >&2; exit 1
  fi
  gzip -t "$partial"; mv "$partial" "$final"; echo "dumped $final"
