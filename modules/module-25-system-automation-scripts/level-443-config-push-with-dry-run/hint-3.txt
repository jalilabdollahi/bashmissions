  [[ -r $local_file ]] || { echo "cannot read $local_file" >&2; exit 1; }
  [[ -s $local_file ]] || { echo "refusing to push empty file: $local_file" >&2; exit 1; }
  while IFS= read -r host; do
    [[ -z $host || $host == \#* ]] && continue
    if (( dry )); then echo "would push $local_file to $host:$remote_path"
    elif scp -q -o BatchMode=yes "$local_file" "$host:$remote_path"; then ...pushed...
    else ...FAILED...
    fi
  done < "$hostfile"
