  [[ -z ${LF_TOKEN:-} ]] && { echo "LF_TOKEN is not set" >&2; exit 2; }
  body_file=$(mktemp)
  trap 'rm -f "$body_file"' EXIT
  code=$(curl -s -o "$body_file" -w '%{http_code}' \
    -H "Authorization: Bearer $LF_TOKEN" URL) || code=000
  case $code in
    200) ... ;;
    401) ... ;;
    *)   ... ;;
  esac

The error messages go to stderr. Never put $LF_TOKEN inside an echo or printf.
