if out=$(git describe --tags --long --dirty --match 'v[0-9]*' 2>/dev/null); then :; else
  # fallback with git rev-parse --short HEAD, then exit 0
fi

re='^v(.+)-([0-9]+)-g([0-9a-f]+)(-dirty)?$'
[[ $out =~ $re ]] || { echo "cannot parse describe output: $out" >&2; exit 1; }
tag=${BASH_REMATCH[1]} distance=${BASH_REMATCH[2]} hash=${BASH_REMATCH[3]} dirty=${BASH_REMATCH[4]}

# also check that $tag starts like 1.2.3
# distance 0 and empty $dirty -> version=$tag, otherwise version=$tag-dev.$distance+g$hash plus ".dirty" if dirty
