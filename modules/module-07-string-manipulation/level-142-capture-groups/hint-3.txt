  re='"([A-Z]+) ([^ ]+) HTTP/[0-9.]+" ([0-9]{3}) '
  while IFS= read -r line || [[ -n $line ]]; do
    if [[ $line =~ $re ]]; then
      echo "${BASH_REMATCH[3]} ${BASH_REMATCH[1]} ${BASH_REMATCH[2]}"
    else
      echo "skipped: $line"
    fi
  done

The regex covers the quoted request, the protocol (not captured) and the status. Read group 1 = method, 2 = path, 3 = status.
