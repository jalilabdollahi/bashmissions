  level_num() { case $1 in DEBUG) echo 0 ;; INFO) ... ;; WARN) ... ;; ERROR) ... ;; esac; }

  log() {
    local level=$1
    shift
    if (( $(level_num "$level") >= $(level_num "${LOG_LEVEL:-INFO}") )); then
      echo "[$level] $*" >&2
    fi
  }
